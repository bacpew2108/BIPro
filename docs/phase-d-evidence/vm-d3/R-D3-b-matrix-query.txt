// DAX Query
DEFINE
	VAR __DS0FilterTable = 
		FILTER(
			KEEPFILTERS(VALUES('Dim_Date'[year])),
			NOT('Dim_Date'[year] IN {9999})
		)

	VAR __DS0FilterTable2 = 
		FILTER(
			KEEPFILTERS(VALUES('Dim_Client'[client_region])),
			NOT('Dim_Client'[client_region] IN {"Unknown"})
		)

	VAR __DS0Core = 
		SUMMARIZECOLUMNS(
			ROLLUPADDISSUBTOTAL(
				ROLLUPGROUP('Dim_Date'[month_name], 'Dim_Date'[month_num]), "IsGrandTotalRowTotal"
			),
			__DS0FilterTable,
			__DS0FilterTable2,
			"Total_Credit_Inflow", 'Fact_Transaction'[Total_Credit_Inflow]
		)

	VAR __DS0PrimaryWindowed = 
		TOPN(
			502,
			__DS0Core,
			[IsGrandTotalRowTotal],
			0,
			'Dim_Date'[month_num],
			1,
			'Dim_Date'[month_name],
			1
		)

EVALUATE
	__DS0PrimaryWindowed

ORDER BY
	[IsGrandTotalRowTotal] DESC, 'Dim_Date'[month_num], 'Dim_Date'[month_name]
