// DAX Query
DEFINE
	VAR __DS0FilterTable = 
		FILTER(
			KEEPFILTERS(VALUES('Fact_Transaction'[trans_hour])),
			NOT('Fact_Transaction'[trans_hour] IN {17})
		)

	VAR __DS0FilterTable2 = 
		FILTER(
			KEEPFILTERS(VALUES('Dim_Date'[year])),
			NOT('Dim_Date'[year] IN {9999})
		)

	VAR __DS0FilterTable3 = 
		FILTER(
			KEEPFILTERS(VALUES('Dim_Client'[client_region])),
			NOT('Dim_Client'[client_region] IN {"Unknown"})
		)

	VAR __DS0Core = 
		SELECTCOLUMNS(
			KEEPFILTERS(
				FILTER(
					KEEPFILTERS(
						SUMMARIZECOLUMNS(
							'Fact_Transaction'[trans_hour],
							'Dim_Date'[day_name],
							'Dim_Date'[day_of_week],
							__DS0FilterTable,
							__DS0FilterTable2,
							__DS0FilterTable3,
							"CountRowsFact_Transaction", COUNTROWS('Fact_Transaction')
						)
					),
					OR(
						OR(
							NOT(ISBLANK('Fact_Transaction'[trans_hour])),
							NOT(ISBLANK('Dim_Date'[day_name]))
						),
						NOT(ISBLANK('Dim_Date'[day_of_week]))
					)
				)
			),
			"'Fact_Transaction'[trans_hour]", 'Fact_Transaction'[trans_hour],
			"'Dim_Date'[day_name]", 'Dim_Date'[day_name],
			"'Dim_Date'[day_of_week]", 'Dim_Date'[day_of_week]
		)

	VAR __DS0PrimaryWindowed = 
		TOPN(
			101,
			SUMMARIZE(__DS0Core, 'Fact_Transaction'[trans_hour]),
			'Fact_Transaction'[trans_hour],
			1
		)

	VAR __DS0SecondaryBase = 
		SUMMARIZE(__DS0Core, 'Dim_Date'[day_name], 'Dim_Date'[day_of_week])

	VAR __DS0Secondary = 
		TOPN(102, __DS0SecondaryBase, 'Dim_Date'[day_of_week], 1, 'Dim_Date'[day_name], 1)

	VAR __DS0BodyLimited = 
		NATURALLEFTOUTERJOIN(
			__DS0PrimaryWindowed,
			SUBSTITUTEWITHINDEX(
				__DS0Core,
				"ColumnIndex",
				__DS0Secondary,
				'Dim_Date'[day_of_week],
				ASC,
				'Dim_Date'[day_name],
				ASC
			)
		)

EVALUATE
	__DS0Secondary

ORDER BY
	'Dim_Date'[day_of_week], 'Dim_Date'[day_name]

EVALUATE
	__DS0BodyLimited

ORDER BY
	'Fact_Transaction'[trans_hour], [ColumnIndex]
